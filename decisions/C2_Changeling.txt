# =====================================================================
# Celestia 2 - Changeling decisions
#
#  CHN_the_one_queen  ("The One Queen Of The Changeling Lands")
#    Pan-changeling unification, modelled on the vanilla German/Italian
#    formation decisions. A changeling hive whose capital lies in Equus
#    and which controls the five great hives -- Vesalipolis (157),
#    Ditrysium (126), Vraks (177), Lyctida (1328) and Soryth (1742) --
#    either directly, through its subjects, or through sphere members,
#    and which has researched Nationalism & Imperialism, may crown its
#    queen the One Queen Of The Changeling Lands.
#
#    Effects:
#      * the country becomes CHN (change_tag preserves the former hive's
#        primary culture, so a boreal hive yields a boreal CHN with
#        brackish accepted, and a brackish hive yields the reverse;
#        CHN's own default is boreal primary / brackish accepted);
#      * every changeling subject and sphere member on Equus is absorbed;
#      * the remaining independent Equus hives receive event 9991460
#        ("Envoys from our kin") and are annexed into CHN if they accept;
#      * +50 prestige.
#
#    CHN is a formable tag: it owns no land at start. Its cores are
#    pre-placed on every changeling-majority (>50% pop) province in
#    Equus (see the province history files).
# =====================================================================

political_decisions = {
	CHN_the_one_queen = {
		potential = {
			culture_group = changeling
			continent = equus
			NOT = { tag = CHN }
		}
		allow = {
			nationalism_n_imperialism = 1

			# Vesalipolis
			OR = {
				owns = 157
				157 = { owner = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } } }
			}
			# Ditrysium
			OR = {
				owns = 126
				126 = { owner = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } } }
			}
			# Vraks
			OR = {
				owns = 177
				177 = { owner = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } } }
			}
			# Lyctida
			OR = {
				owns = 1328
				1328 = { owner = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } } }
			}
			# Soryth
			OR = {
				owns = 1742
				1742 = { owner = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } } }
			}
		}
		effect = {
			change_tag = CHN

			if = {
				limit = { primary_culture = boreal }
				add_accepted_culture = brackish
			}
			if = {
				limit = { primary_culture = brackish }
				add_accepted_culture = boreal
			}

			any_country = {
				limit = {
					OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS }
					culture_group = changeling
					continent = equus
				}
				annex_to = THIS
			}

			# Hives that are still independent - not our subjects, substates
			# or sphere members - are not annexed outright: each receives an
			# invitation to join the union of its own will (event 9991460).
			any_country = {
				limit = {
					culture_group = changeling
					continent = equus
					NOT = { tag = CHN }
					NOT = { OR = { vassal_of = THIS substate_of = THIS in_sphere = THIS } }
				}
				country_event = { id = 9991460 days = 0 }
			}

			prestige = 50
		}
		ai_will_do = {
			factor = 1
		}
	}

	# "The Acornage Boundary Agreement" - after unifying the changelings, CHN
	# settles its southern border with Equestria: CHN takes the Wild Lands
	# (state_157) and Svarm (state_155); Equestria takes Fort Acorn (state_11),
	# Fairflanks-Whitebell (state_13) and West/East Yeahcon (state_14/state_20)
	# and gains cores on them. Fort Acorn is renamed Acornage.
	CHN_acornage_boundary = {
		potential = {
			tag = CHN
			NOT = { has_country_flag = CHN_acornage_boundary_done }
		}
		allow = {
			war = no
			EQS = { war = no }
			any_neighbor_country = { tag = EQS }
			relation = { who = EQS value >= 0 }
		}
		effect = {
			# Wild Lands and Svarm go to the Changelings.
			195 = { secede_province = THIS }
			217 = { secede_province = THIS }
			220 = { secede_province = THIS }
			223 = { secede_province = THIS }
			257 = { secede_province = THIS }
			261 = { secede_province = THIS }
			292 = { secede_province = THIS }

			# Fort Acorn goes to Equestria; its town and state are renamed.
			250 = {
				secede_province = EQS
				add_core = EQS
				change_province_name = "Acornage"
				change_region_name = "Acornage"
			}
			# Fairflanks-Whitebell (state_13)
			215 = { secede_province = EQS add_core = EQS }
			234 = { secede_province = EQS add_core = EQS }
			221 = { secede_province = EQS add_core = EQS }
			265 = { secede_province = EQS add_core = EQS }
			# The rest of Fort Acorn (state_11)
			272 = { secede_province = EQS add_core = EQS }
			231 = { secede_province = EQS add_core = EQS }
			262 = { secede_province = EQS add_core = EQS }
			# West Yeahcon (state_14)
			280 = { secede_province = EQS add_core = EQS }
			245 = { secede_province = EQS add_core = EQS }
			275 = { secede_province = EQS add_core = EQS }
			255 = { secede_province = EQS add_core = EQS }
			269 = { secede_province = EQS add_core = EQS }
			# East Yeahcon (state_20)
			312 = { secede_province = EQS add_core = EQS }
			290 = { secede_province = EQS add_core = EQS }
			626 = { secede_province = EQS add_core = EQS }
			314 = { secede_province = EQS add_core = EQS }
			305 = { secede_province = EQS add_core = EQS }

			set_country_flag = CHN_acornage_boundary_done
		}
		ai_will_do = {
			factor = 1
		}
	}
}
