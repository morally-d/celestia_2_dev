########################################################################
#  Celestia 2 - Gold Rush Events
#
#  Mimics vanilla events/Goods.txt (province_event 1100, the one that
#  turns Caozhou and friends into precious_metal):
#  instead of setting trade_goods in history/provinces/*, the province's
#  main RGO is switched to precious_metal by a province_event after the
#  game has already started.  This deliberately keeps the new gold out of
#  the pre-start economy simulation, and simply injects extra coin into
#  the world market once play begins.
#
#  Event ids reserved for this file: 9991300 - 9991399
#  When adding more provinces, continue at the next free id and add the
#  matching EVTNAME / EVTDESC / EVTOPTA keys to:
#    localisation/C2GoldRush.csv            (English)
#    assets/localisation/zh-CN/zzzz-C2-GoldRush.csv   (Chinese)
########################################################################

province_event = {
	id = 9991300

	trigger = {
		year = 932
		province_id = 772			# The Mountain
		not = { trade_goods = precious_metal }
	}

	mean_time_to_happen = {
		months = 12
	}

	fire_only_once = yes
	title = "EVTNAME9991300"
	desc = "EVTDESC9991300"

	option = {
		name = "EVTOPTA9991300"
		trade_goods = precious_metal
		add_province_modifier = {
			name = gold_rush
			duration = 730
		}
	}
}

	}

		name = "EVTOPTA9991307"

			duration = 365
		}
	}
