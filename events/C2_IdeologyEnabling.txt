# =====================================================================
# Celestia 2 - enable ideologies across the 932-1032 timeline.
#
# Why: in PA an ideology's `date` does NOT auto-enable it. Vanilla V2 did
# that from events/IdeologyEnabling.txt, which this mod's
# `replace_path = "events"` removed -- so socialist / communist / anarcho_liberal /
# fascist stayed disabled forever (only 3 of 7 ideologies existed).
#
# Fired from common/on_actions.txt -> on_yearly_pulse (every October, per nation).
# `enable_ideology` is global, so the first nation that fires it turns it on for everyone.
# Dates here must match common/ideologies.txt.
# =====================================================================

#####################
# Enable Socialists (from 944)
#####################
country_event = {
	id = 1000
	title = "EVTNAME1000"
	desc = "EVTDESC1000"

	trigger = {
		NOT = { is_ideology_enabled = socialist }
		year = 944
	}

	fire_only_once = yes
	immediate = { enable_ideology = socialist }

	option = {
		name = "EVTOPTA1000"
	}
}
